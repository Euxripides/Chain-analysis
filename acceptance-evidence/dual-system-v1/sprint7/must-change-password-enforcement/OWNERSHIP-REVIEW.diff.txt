+// Session workspace staleness guard (sprint7 C1-RACE, defence in depth).
+//
+// Measured defect: a state-changing request that was authenticated while the session was in
+// workspace A could still commit into workspace A after the same session had been switched
+// to workspace B -- the principal is resolved once, at the start of the request, while the
+// request body is still being uploaded. The security-critical variant (membership revoked in
+// A) is already fail closed by the feature-level membership re-check; this guard closes the
+// remaining window, where the SESSION workspace itself goes stale mid-request.
+//
+// Contract: for state-changing requests only (never GET/HEAD/OPTIONS) the guard compares
+// principal.workspace_id with the CURRENT workspace of the live session row, read with one
+// primary-key lookup. A missing or revoked session row, a different workspace, and any
+// database error all refuse with HTTP 409 and the stable code
+// AUTHZ_WORKSPACE_SCOPE_STALE. Safe methods never touch the database.
+// ---------------------------------------------------------------------------
+
+// ErrorAUTHZWorkspaceScopeStale is the stable refusal code (HTTP 409) for a state-changing
+// request whose session workspace no longer matches the live session row.
+const ErrorAUTHZWorkspaceScopeStale = "AUTHZ_WORKSPACE_SCOPE_STALE"

+// workspaceScopeStaleMessage is the user-facing text of that refusal. It names no object, no

+// 使"下载成功必须留下服务端审计"这类关键路径能被 httptest 逐字段断言（D-R10V9-01）。
+type auditSink interface {
+	Write(ctx context.Context, e audit.Event) error
+}
+
+// emitAudit 统一审计出口。写入失败**不静默**：沿用既有关键操作审计策略 —— 记录 Error 日志
+// 并返回错误，由调用方决定是 fail closed 还是业务成功 + 明确审计告警。绝不吞掉错误。
+//
+// D-B2RAW-02 (2026-09-13 security review, real defect): auditRequestDetached uses
+// context.WithoutCancel to survive client disconnect, which ALSO drops the request deadline.
+// audit.Writer.Write then runs Begin/INSERT/Commit with that context, while production
+// PostgreSQL has statement_timeout=0, every role has an empty rolconfig and the pgxpool sets
+// no per-query timeout -- so this audit write was completely UNBOUNDED. If the database stalls
+// (lock wait, saturated pool) the download handler goroutine and its pooled connection stay
+// pinned forever; repeated downloads can exhaust the pool and turn an audit-reliability fix
+// into an availability defect. Bound every audit write that has no deadline; contexts that
+// already carry an earlier deadline are left untouched.
+func (s *Server) emitAudit(ctx context.Context, ev audit.Event) error {
+	if s.auditSink == nil && s.audit == nil {
+		return nil
+	}
+	ctx, cancel := auditBoundedContext(ctx)
+	defer cancel()
+	if s.auditSink != nil {
+		return s.auditSink.Write(ctx, ev)
+	}
+	return s.audit.Write(ctx, ev)
+}
+
+// auditWriteTimeout bounds a single audit write.
+// It must stay a var (not a const): the regression test lowers it to prove deterministically
+// that an unbounded write really is cut off.
+var auditWriteTimeout = 5 * time.Second
+
+// auditBoundedContext clamps an audit write to the SMALLER of the caller's own deadline and
+// now+auditWriteTimeout, returning the matching cancel.
+//
+// D-B2RAW-03 (2026-09-13 review, second face of D-B2RAW-02): the first fix only added a bound
+// when the caller had NO deadline, so any caller carrying a deadline LONGER than the bound
+// (a request/background context with its own multi-minute or one-hour timeout) silently kept an
+// effectively unbounded audit write, and TestAuditBoundedContextAddsAndPreservesDeadline pinned
+// that hole with a one-hour parent. The safety intent is an upper bound on the write, so a later
+// deadline is now clamped down to now+auditWriteTimeout; an EARLIER deadline is still preserved
+// exactly, because shortening the caller's own semantics is never allowed either.
+func auditBoundedContext(ctx context.Context) (context.Context, context.CancelFunc) {
+	limit := time.Now().Add(auditWriteTimeout)
+	if dl, ok := ctx.Deadline(); ok && !dl.After(limit) {
+		return ctx, func() {}
+	}
+	return context.WithDeadline(ctx, limit)
+}

+# ---------------------------------------------------------------------------
+# Sprint 7 API-layer authorization guards.
+#
+# These are enforced by the Go middleware layer (backend/internal/api), not by an individual
+# endpoint, so they apply to every business operation documented in user-v2.yaml,
+# admin-v1.yaml and auth-v1.yaml. Both answer the standard ErrorEnvelope above.
+# ---------------------------------------------------------------------------

+# Server-side enforcement of identities.must_change_password (sprint7 C1-SEC).
+# A session whose identity still carries must_change_password = true is refused on every
+# business surface BEFORE any business logic runs. The self-service surface stays reachable so
+# the holder can repair the state, and health/metrics/docs are untouched.
+MustChangePasswordGuard:
+  refused_code: AUTHN_PASSWORD_CHANGE_REQUIRED
+  refused_status: 403
+  guarded_surfaces:
+    - /api/v1/
+    - /api/v2/
+    - /api/admin/v1/
+  fail_closed: true
+  fail_closed_note: >-
+    Any /api/ path that is neither allowlisted nor part of the self-service auth surface is
+    guarded, so a future surface is refused by default without a code change.
+  reachable_while_flagged:
+    - GET /api/auth/v1/session
+    - POST /api/auth/v1/logout
+    - POST /api/auth/v1/password/change
+    - GET /api/auth/v1/methods
+  unchanged_behaviour:
+    - every other /api/auth/v1 route keeps its previous behaviour
+    - /health/live, /health/ready, /metrics, /openapi.json, /docs, /redoc are outside the guard
+  audit: >-
+    action authorization.denied, category authorization, result deny, http_status 403,
+    error_code AUTHN_PASSWORD_CHANGE_REQUIRED, deny_reason must_change_password, plus the
+    route and the request id. No password, password hash or session token is ever recorded.
+  response:
+    description: >-
+      The authenticated identity must change its password first. details.reason is
+      must_change_password and details.password_change_endpoint names the self-service route
+      that clears the flag.
+    content:
+      application/json:
+        schema:
+          $ref: '#/ErrorEnvelope'

+# Session workspace staleness guard (sprint7 C1-RACE, defence in depth).
+# A state-changing request authenticated while the session was in workspace A must not commit
